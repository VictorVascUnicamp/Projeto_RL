# generate_experiments.py
import os
import json
import itertools
import copy

# 1. Cria a pasta isolada
pasta_destino = "test_configs"
os.makedirs(pasta_destino, exist_ok=True)

# 2. Carrega o Molde
with open("config.json", "r") as f:
    base_config = json.load(f)

# --- A LISTA DE GRADES MÁGICA ---
# Cada dicionário é um sub-experimento isolado.
grades_de_teste =[
    # Modelos Clássicos
    # {
    #     "env": ["CartPole", "Reacher2D"],
    #     "agent":["PPO", "DQN","Random"],
    #     "mode": ["episodic", "continuous"]
    # },
    # AHaH com 3 Bias
    {
        "env":["Reacher2D"],
        "agent": ["AHaH"],
        "mode": ["episodic", "continuous"],
        "bias":["A","B","C"],
        "temporal": [False],
        "encoder": ["adaptive"]
    },
    # Walker Clássico
    # {
    #     "env": ["BipedalWalker"],
    #     "agent":["PPO", "DQN","Random"],
    #     "mode": ["episodic"]
    # },
    # Encoder Fixo
    {
        "env": ["Reacher2D"],
        "agent": ["AHaH"],
        "mode": ["episodic", "continuous"],
        "bias": ["B"],           
        "temporal": [False],
        "encoder": ["fixed"]
    },
    # Encoder Temporal
    # {
    #     "env": ["CartPole"],
    #     "agent": ["AHaH"],
    #     "mode": ["episodic", "continuous"],
    #     "bias": ["B"],           
    #     "temporal": [True],
    #     "encoder": ["adaptive"]
    # }

]

contador = 0

for grade in grades_de_teste:
    # Prepara as listas (com valores padrão se não existirem na grade)
    envs = grade.get("env", ["CartPole"])
    agents = grade.get("agent",["AHaH"])
    modes = grade.get("mode", ["episodic"])
    biases = grade.get("bias", [None])
    temporais = grade.get("temporal", [None])
    encoders = grade.get("encoder", [None])

    # Cria todas as combinações APENAS para os limites desta grade
    for env, agent, mode, bias, temporal, encoder in itertools.product(envs, agents, modes, biases, temporais, encoders):
        # Cria uma cópia limpa do molde
        cfg = copy.deepcopy(base_config)
        
        # --- APLICAÇÃO GERAL ---
        cfg["selection"]["environment"] = env
        cfg["selection"]["agent"] = agent
        cfg["training"]["learning_mode"] = mode
        
        # --- APLICAÇÃO DE REGRAS DE NEGÓCIO E SEGURANÇA ---
        env_dict = cfg["definitions"]["environments"][env]
        
        if mode == "continuous":
            cfg["training"]["episodes"] = 1            
            
            if env == "CartPole":
                env_dict["use_reward_shaping"] = True
                cfg["training"]["continuous_target_count"] = 1000
            elif env == "Reacher2D":
                cfg["training"]["continuous_target_count"] = 100

            if agent == "PPO":
                env_dict["max_steps"] = 500000
            elif agent == "Random":
                env_dict["max_steps"] = 10000
            else:
                env_dict["max_steps"] = 100000
                
        elif mode == "episodic":
            if env == "CartPole":
                env_dict["max_steps"] = 500
                if agent == "AHaH":
                    env_dict["use_reward_shaping"] = True
            elif env == "Reacher2D":
                env_dict["max_steps"] = 300
            elif env == "BipedalWalker":
                env_dict["max_steps"] = 1600

            if agent == "PPO":
                cfg["training"]["episodes"] = 5000
            elif agent == "Random":
                cfg["training"]["episodes"] = 100
            else:
                cfg["training"]["episodes"] = 1000

        # Monta o nome do arquivo dinamicamente (ex: CartPole_AHaH_cont)
        # [:4] pega só as 4 primeiras letras (epis / cont) para o nome não ficar gigante
        nome_partes = [env, agent, mode[:4]]
        
        # --- APLICAÇÃO ESPECÍFICA DO AHaH ---
        if agent == "AHaH":
            cfg["definitions"]["agents"]["AHaH"]["bias_update_mode"] = bias
            cfg["definitions"]["agents"]["AHaH"]["use_temporal_encoder"] = temporal
            cfg["definitions"]["agents"]["AHaH"]["encoder_type"] = encoder
            cfg["training"]["freeze"] = False
            
            nome_partes.append(f"Bias{bias}")
            if temporal:
                nome_partes.append("Temp")
            if encoder == "fixed":
                nome_partes.append("Fixed")
        else:
            cfg["training"]["freeze"] = True
            
        # 5. Salva o arquivo final
        nome_arquivo = "exp_" + "_".join(nome_partes) + ".json"
        caminho_final = os.path.join(pasta_destino, nome_arquivo)
        
        with open(caminho_final, "w") as f:
            json.dump(cfg, f, indent=2)
            
        print(f"Gerado: {nome_arquivo}")
        contador += 1

print(f"\n[SUCESSO] {contador} arquivos de configuração criados na pasta '{pasta_destino}'!")
